cmake_minimum_required(VERSION 3.15)

project(Spectrum VERSION 1.0.0)

juce_add_plugin(Spectrum
    COMPANY_NAME "Mugsend"
    PLUGIN_MANUFACTURER_CODE MugS # Must be exactly 4 characters
    PLUGIN_CODE Spec              # Must be exactly 4 characters
    FORMATS VST3 AU Standalone    # The formats you want to build
    PRODUCT_NAME "Spectrum"
    
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    
    COPY_PLUGIN_AFTER_BUILD TRUE
)

target_sources(Spectrum PRIVATE
    Source/PluginProcessor.cpp
    Source/PluginProcessor.h
    Source/PluginEditor.cpp
    Source/PluginEditor.h
    Source/MagnitudeMeter.cpp
    Source/MagnitudeMeter.h
    Source/CustomLookAndFeel.h
)

target_compile_definitions(Spectrum PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
)

target_link_libraries(Spectrum PRIVATE
    juce::juce_audio_utils
    juce::juce_dsp         
    juce::juce_gui_extra
)
